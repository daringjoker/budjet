# Budjet Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a personal financial management app that ingests bank transaction emails via IMAP IDLE, parses them with a local LLM (node-llama-cpp / Qwen2.5-3B), stores data in SQLite via Drizzle ORM, and exposes everything through a tRPC API consumed by a React dashboard and a CLI.

**Architecture:** Single Node.js process — main thread runs IMAP IDLE + email queue + tRPC HTTP server; 1–2 worker threads run the LLM pipeline. Monorepo (pnpm workspaces + Turborepo) with four packages: `shared`, `backend`, `cli`, `frontend`.

**Tech Stack:** Node.js 20+, TypeScript 5, pnpm workspaces, Turborepo, imapflow, node-llama-cpp, Drizzle ORM + better-sqlite3, tRPC v11, Hono, commander.js, React + Vite, Recharts, Vitest

```typescript
import { EventEmitter } from 'events'
import { ImapFlow, type ImapFlowOptions } from 'imapflow'

export interface EmailEvent {
  messageId: string
  sender: string
  subject: string
  bodyText: string
  receivedAt: Date
}

export class ImapWatcher extends EventEmitter {
  private client: ImapFlow
  private reconnectDelay = 1000
  private stopped = false

  constructor(
    private readonly cfg: { host: string; user: string; password: string; port: number },
    clientOverride?: ImapFlow
  ) {
    super()
    this.client = clientOverride ?? new ImapFlow({
      host: cfg.host,
      port: cfg.port,
      secure: true,
      auth: { user: cfg.user, pass: cfg.password },
      logger: false,
    } as ImapFlowOptions)
  }

  async connect() {
    await this.client.connect()
    await this.client.mailboxOpen('INBOX')

    this.client.on('exists', async (data: { count: number; prevCount: number }) => {
      if (data.count > data.prevCount) {
        await this.fetchUnseen()
      }
    })

    this.client.on('error', (err: Error) => {
      if (!this.stopped) this.scheduleReconnect()
    })

    await this.client.idle()
  }

  private async fetchUnseen() {
    try {
      for await (const msg of this.client.fetch('1:*', {
        uid: true,
        envelope: true,
        bodyParts: ['TEXT'],
        flags: true,
      })) {
        if (msg.flags?.has('\\Seen')) continue
        const bodyBuffer = msg.bodyParts?.get('TEXT')
        const bodyText = bodyBuffer ? bodyBuffer.toString('utf-8') : ''
        const event: EmailEvent = {
          messageId: msg.envelope?.messageId ?? `${msg.uid}`,
          sender: msg.envelope?.from?.[0]?.address ?? '',
          subject: msg.envelope?.subject ?? '',
          bodyText,
          receivedAt: msg.envelope?.date ?? new Date(),
        }
        this.emit('email', event)
      }
    } catch (err) {
      this.emit('error', err)
    }
  }

  private scheduleReconnect() {
    setTimeout(async () => {
      try {
        this.reconnectDelay = Math.min(this.reconnectDelay * 2 + Math.random() * 1000, 60_000)
        await this.connect()
        this.reconnectDelay = 1000
      } catch {
        this.scheduleReconnect()
      }
    }, this.reconnectDelay)
  }

  async stop() {
    this.stopped = true
    await this.client.logout()
  }
}
```

```typescript
import { EventEmitter } from 'events'
import type { EmailEvent } from '../imap/watcher.js'

export class EmailQueue extends EventEmitter {
  private queue: EmailEvent[] = []
  private consumer: ((e: EmailEvent) => Promise<void>) | null = null
  private processing = false
  private readonly maxSize: number

  constructor({ maxSize = 100 }: { maxSize?: number } = {}) {
    super()
    this.maxSize = maxSize
  }

  push(email: EmailEvent) {
    if (this.queue.length >= this.maxSize) {
      const dropped = this.queue.shift()!
      this.emit('dropped', dropped)
    }
    this.queue.push(email)
    this.drain()
  }

  consume(fn: (e: EmailEvent) => Promise<void>) {
    this.consumer = fn
    this.drain()
  }

  private async drain() {
    if (this.processing || !this.consumer || this.queue.length === 0) return
    this.processing = true
    while (this.queue.length > 0 && this.consumer) {
      const email = this.queue.shift()!
      try {
        await this.consumer(email)
      } catch (err) {
        this.emit('error', err)
      }
    }
    this.processing = false
  }
}
```

```typescript
import { Worker } from 'worker_threads'
import { randomUUID } from 'crypto'
import { fileURLToPath } from 'url'
import { resolve, dirname } from 'path'

export interface LLMJob {
  jobId: string
  type: 'classify' | 'parse'
  payload: {
    subject: string
    body: string
    bankName: string
  }
}

export interface LLMResult {
  jobId: string
  result: unknown
  error?: string
}

type WorkerFactory = (path: string, options: object) => Worker

export class LLMPool {
  private workers: Worker[] = []
  private queue: Array<{ job: LLMJob; resolve: (r: unknown) => void; reject: (e: Error) => void }> = []
  private idle: Worker[] = []
  private pending = new Map<string, { resolve: (r: unknown) => void; reject: (e: Error) => void }>()

  constructor(
    private readonly cfg: { workerCount: number; modelPath: string },
    private readonly factory?: WorkerFactory
  ) {}

  async init() {
    const workerPath = resolve(
      dirname(fileURLToPath(import.meta.url)),
      'worker.js'
    )
    for (let i = 0; i < this.cfg.workerCount; i++) {
      const worker = this.factory
        ? this.factory(workerPath, { workerData: { modelPath: this.cfg.modelPath } })
        : new Worker(workerPath, { workerData: { modelPath: this.cfg.modelPath } })

      worker.on('message', (msg: LLMResult) => {
        const pending = this.pending.get(msg.jobId)
        if (!pending) return
        this.pending.delete(msg.jobId)
        if (msg.error) pending.reject(new Error(msg.error))
        else pending.resolve(msg.result)
        this.idle.push(worker)
        this.drain()
      })

      worker.on('error', (err) => {
        for (const [, p] of this.pending) p.reject(err)
        this.pending.clear()
      })

      this.idle.push(worker)
    }
  }

  run(job: LLMJob): Promise<unknown> {
    return new Promise((resolve, reject) => {
      this.queue.push({ job, resolve, reject })
      this.drain()
    })
  }

  private drain() {
    while (this.queue.length > 0 && this.idle.length > 0) {
      const worker = this.idle.pop()!
      const { job, resolve, reject } = this.queue.shift()!
      this.pending.set(job.jobId, { resolve, reject })
      worker.postMessage(job)
    }
  }

  async shutdown() {
    await Promise.all(this.workers.map(w => w.terminate()))
  }
}
```

```typescript
import { randomUUID } from 'crypto'
import { eq } from 'drizzle-orm'
import { rawEmails, transactions, accounts } from '@budjet/shared/schema'
import type { Db } from '../db/index.js'
import type { LLMPool } from '../llm/pool.js'
import type { EmailEvent } from '../imap/watcher.js'
import { detectRecurring } from './recurring-detector.js'

interface BankEntry { name: string; currency: string; accountId?: number }

export class Processor {
  constructor(
    private readonly db: Db,
    private readonly knownSenders: Map<string, BankEntry>,
    private readonly pool: LLMPool,
    private readonly confidenceThreshold: number
  ) {}

  async process(email: EmailEvent) {
    const bank = this.knownSenders.get(email.sender.toLowerCase())

    if (!bank) {
      await this.db.insert(rawEmails).values({
        messageId: email.messageId,
        sender: email.sender,
        subject: email.subject,
        bodyText: email.bodyText,
        receivedAt: email.receivedAt,
        status: 'unknown',
      }).onConflictDoNothing()
      return
    }

    const [savedEmail] = await this.db.insert(rawEmails).values({
      messageId: email.messageId,
      sender: email.sender,
      subject: email.subject,
      bodyText: email.bodyText,
      receivedAt: email.receivedAt,
      status: 'unknown',
      accountId: bank.accountId,
    }).onConflictDoNothing().returning()

    if (!savedEmail) return // duplicate

    try {
      const classification = await this.pool.run({
        jobId: randomUUID(),
        type: 'classify',
        payload: { subject: email.subject, body: email.bodyText, bankName: bank.name },
      }) as { type: string; confidence: number }

      if (classification.type !== 'transaction') {
        await this.db.update(rawEmails).set({ status: 'unknown' }).where(eq(rawEmails.id, savedEmail.id))
        return
      }

      const parsed = await this.pool.run({
        jobId: randomUUID(),
        type: 'parse',
        payload: { subject: email.subject, body: email.bodyText, bankName: bank.name },
      }) as { amount: number; type: string; description: string; currency: string; balance_after?: number | null; confidence: number; date?: string }

      const txDate = parsed.date ? new Date(parsed.date) : email.receivedAt

      const [txn] = await this.db.insert(transactions).values({
        accountId: bank.accountId!,
        date: txDate,
        amount: parsed.amount,
        type: parsed.type as 'debit' | 'credit',
        description: parsed.description,
        currency: parsed.currency,
        balanceAfter: parsed.balance_after ?? null,
        source: 'email',
        rawEmailId: savedEmail.id,
        llmConfidence: parsed.confidence,
        createdAt: new Date(),
      }).returning()

      await this.db.update(rawEmails).set({ status: 'processed' }).where(eq(rawEmails.id, savedEmail.id))

      if (parsed.confidence >= this.confidenceThreshold) {
        await detectRecurring(this.db, txn)
      }
    } catch {
      await this.db.update(rawEmails).set({ status: 'failed' }).where(eq(rawEmails.id, savedEmail.id))
    }
  }
}
```

```typescript
import { and, eq, gte, sql } from 'drizzle-orm'
import { transactions, recurringTransactions } from '@budjet/shared/schema'
import type { Db } from '../db/index.js'
import type { Transaction } from '@budjet/shared/schema'

const LOOKBACK_DAYS = 90
const MIN_OCCURRENCES = 3
const AMOUNT_TOLERANCE = 0.1 // 10%

export async function detectRecurring(db: Db, txn: Transaction) {
  if (!txn.description) return

  const since = new Date(Date.now() - LOOKBACK_DAYS * 86_400_000)

  const similar = await db
    .select()
    .from(transactions)
    .where(
      and(
        eq(transactions.accountId, txn.accountId),
        eq(transactions.type, txn.type),
        gte(transactions.date, since),
        sql`lower(${transactions.description}) = lower(${txn.description})`
      )
    )
    .orderBy(transactions.date)

  if (similar.length < MIN_OCCURRENCES) return

  const amounts = similar.map(t => t.amount)
  const avgAmount = amounts.reduce((a, b) => a + b, 0) / amounts.length
  const withinTolerance = amounts.every(a => Math.abs(a - avgAmount) / avgAmount <= AMOUNT_TOLERANCE)
  if (!withinTolerance) return

  // Estimate interval in days between occurrences
  const intervals: number[] = []
  for (let i = 1; i < similar.length; i++) {
    const diff = (similar[i].date.getTime() - similar[i - 1].date.getTime()) / 86_400_000
    intervals.push(diff)
  }
  const avgInterval = intervals.reduce((a, b) => a + b, 0) / intervals.length

  let frequency: 'weekly' | 'biweekly' | 'monthly' | 'quarterly' | 'yearly' | null = null
  if (avgInterval < 9) frequency = 'weekly'
  else if (avgInterval < 20) frequency = 'biweekly'
  else if (avgInterval < 45) frequency = 'monthly'
  else if (avgInterval < 120) frequency = 'quarterly'
  else if (avgInterval < 400) frequency = 'yearly'

  if (!frequency) return

  const nextDue = new Date(txn.date.getTime() + avgInterval * 86_400_000)

  await db.insert(recurringTransactions).values({
    accountId: txn.accountId,
    name: txn.description,
    amount: avgAmount,
    currency: txn.currency,
    type: txn.type,
    frequency,
    startDate: similar[0].date,
    nextDueDate: nextDue,
    autoDetected: true,
    toleranceAmount: avgAmount * AMOUNT_TOLERANCE,
    toleranceDays: 3,
    status: 'pending_confirmation',
    createdAt: new Date(),
  })
}
```

```typescript
export class Scheduler {
  private timers: NodeJS.Timeout[] = []

  daily(hour: number, fn: () => Promise<void>) {
    const now = new Date()
    const next = new Date(now)
    next.setHours(hour, 0, 0, 0)
    if (next <= now) next.setDate(next.getDate() + 1)
    const delay = next.getTime() - now.getTime()

    const timeout = setTimeout(() => {
      fn().catch(console.error)
      this.daily(hour, fn) // reschedule
    }, delay)

    this.timers.push(timeout)
  }

  stop() {
    this.timers.forEach(t => clearTimeout(t))
    this.timers = []
  }
}
```

```typescript
import { and, eq, gte, lte, sum, avg } from 'drizzle-orm'
import { transactions, predictedTransactions, budgets, budgetAlerts, recurringTransactions } from '@budjet/shared/schema'
import type { Db } from '../db/index.js'

export async function runNightlyForecast(db: Db, lookbackDays: number) {
  const now = new Date()
  const since = new Date(now.getTime() - lookbackDays * 86_400_000)
  const monthStart = new Date(now.getFullYear(), now.getMonth(), 1)
  const monthEnd = new Date(now.getFullYear(), now.getMonth() + 1, 0)

  // 1. Mark missed predicted transactions
  const pendingPast = await db.select().from(predictedTransactions)
    .where(and(eq(predictedTransactions.status, 'pending'), lte(predictedTransactions.predictedDate, now)))
  for (const p of pendingPast) {
    await db.update(predictedTransactions).set({ status: 'missed' }).where(eq(predictedTransactions.id, p.id))
  }

  // 2. Generate recurring predicted entries for next 30 days
  const activeRecurring = await db.select().from(recurringTransactions).where(eq(recurringTransactions.status, 'active'))
  const horizon = new Date(now.getTime() + 30 * 86_400_000)

  for (const r of activeRecurring) {
    let next = r.nextDueDate
    while (next <= horizon) {
      const exists = await db.select({ id: predictedTransactions.id })
        .from(predictedTransactions)
        .where(and(
          eq(predictedTransactions.recurringId, r.id),
          eq(predictedTransactions.status, 'pending'),
          gte(predictedTransactions.predictedDate, new Date(next.getTime() - r.toleranceDays * 86_400_000)),
          lte(predictedTransactions.predictedDate, new Date(next.getTime() + r.toleranceDays * 86_400_000)),
        ))
      if (!exists.length) {
        await db.insert(predictedTransactions).values({
          type: 'recurring',
          recurringId: r.id,
          accountId: r.accountId,
          categoryId: r.categoryId,
          predictedAmount: r.amount,
          predictedDate: next,
          confidence: 0.9,
          basis: 'recurring_pattern',
          status: 'pending',
          createdAt: now,
        })
      }
      next = advanceByFrequency(next, r.frequency)
    }
  }

  // 3. Check budget alerts
  const allBudgets = await db.select().from(budgets)
  for (const budget of allBudgets) {
    const [{ spent }] = await db.select({ spent: sum(transactions.amount) })
      .from(transactions)
      .where(and(
        eq(transactions.type, 'debit'),
        gte(transactions.date, monthStart),
        lte(transactions.date, monthEnd),
        budget.accountId ? eq(transactions.accountId, budget.accountId) : undefined,
      ))
    const spentNum = Number(spent ?? 0)
    const pct = spentNum / budget.amount

    const alerts = await db.select().from(budgetAlerts).where(eq(budgetAlerts.budgetId, budget.id))
    for (const alert of alerts) {
      if (pct >= alert.threshold) {
        const alreadyFired = alert.notifiedAt && alert.notifiedAt > monthStart
        if (!alreadyFired) {
          console.warn(`[BUDGET ALERT] "${budget.name}" is at ${Math.round(pct * 100)}% (threshold: ${alert.threshold * 100}%)`)
          await db.update(budgetAlerts).set({ notifiedAt: now }).where(eq(budgetAlerts.id, alert.id))
        }
      }
    }
  }
}

function advanceByFrequency(date: Date, frequency: string): Date {
  const d = new Date(date)
  switch (frequency) {
    case 'daily': d.setDate(d.getDate() + 1); break
    case 'weekly': d.setDate(d.getDate() + 7); break
    case 'biweekly': d.setDate(d.getDate() + 14); break
    case 'monthly': d.setMonth(d.getMonth() + 1); break
    case 'quarterly': d.setMonth(d.getMonth() + 3); break
    case 'yearly': d.setFullYear(d.getFullYear() + 1); break
  }
  return d
}
```

```typescript
import type { Command } from 'commander'
import type { trpc as TrpcClient } from '../index.js'
import chalk from 'chalk'

export function registerAccountsCommands(program: Command, trpc: typeof TrpcClient) {
  const cmd = program.command('accounts').description('Manage accounts')

  cmd.command('list').description('List all accounts').action(async () => {
    const list = await trpc.accounts.list.query()
    if (!list.length) { console.log('No accounts found.'); return }
    console.log(chalk.bold('\nAccounts:'))
    list.forEach(a => {
      console.log(`  ${chalk.cyan(a.name)} (${a.type}) — ${a.currency} — ID: ${a.id}`)
    })
  })

  cmd.command('add')
    .description('Add a new account')
    .requiredOption('--name <name>', 'Account name')
    .requiredOption('--type <type>', 'Account type: bank|cash|debit_card|credit_card|fund|investment')

    .option('--sender <email>', 'Bank email sender address')

      const created = await trpc.accounts.create.mutate({
        name: opts.name,
        type: opts.type,
        currency: opts.currency,
        bankName: opts.bank,
        emailSender: opts.sender,
        initialBalance: parseFloat(opts.balance),
```

```typescript
    .option('--account <id>', 'Filter by account ID')
    .option('--from <date>', 'From date (ISO)')
    .option('--to <date>', 'To date (ISO)')

    .requiredOption('--amount <amount>', 'Amount')
```

```typescript
import type { Command } from 'commander'
import type { trpc as TrpcClient } from '../index.js'
import chalk from 'chalk'
import * as readline from 'readline/promises'

export function registerReviewCommands(program: Command, trpc: typeof TrpcClient) {
  program.command('review')
    .description('Review low-confidence parsed transactions')
    .action(async () => {
      const list = await trpc.transactions.list.query({ lowConfidenceOnly: true, limit: 20 })
      const lowConf = list.filter(t => t.llmConfidence !== null && t.llmConfidence < 0.85)
      if (!lowConf.length) { console.log(chalk.green('No transactions need review.')); return }

      const rl = readline.createInterface({ input: process.stdin, output: process.stdout })
      console.log(chalk.bold(`\n${lowConf.length} transactions need review:\n`))

      for (const t of lowConf) {
        console.log(`  ID: ${t.id} | ${t.date.toISOString().slice(0, 10)} | ${t.type} ${t.amount} ${t.currency}`)
        console.log(`  Description: ${chalk.yellow(t.description)} (confidence: ${t.llmConfidence?.toFixed(2)})`)
        const newDesc = await rl.question('  New description (Enter to keep): ')
        if (newDesc.trim()) {
          await trpc.transactions.review.mutate({ id: t.id, description: newDesc.trim(), llmConfidence: null })
          console.log(chalk.green('  ✓ Updated\n'))
        } else {
          console.log('  Skipped\n')
        }
      }
      rl.close()
    })
}
```

```typescript
import type { Command } from 'commander'
import type { trpc as TrpcClient } from '../index.js'
import chalk from 'chalk'

export function registerReportCommands(program: Command, trpc: typeof TrpcClient) {
  program.command('report')
    .description('Financial summary report')
    .option('--period <period>', 'this-month | this-week | YYYY-MM', 'this-month')
    .option('--account <id>', 'Filter by account ID')
    .action(async (opts) => {
      const { from, to } = resolvePeriod(opts.period)
      const accountId = opts.account ? parseInt(opts.account) : undefined

      const [summary, breakdown] = await Promise.all([
        trpc.reports.summary.query({ from: from.toISOString(), to: to.toISOString(), accountId }),
        trpc.reports.categoryBreakdown.query({ from: from.toISOString(), to: to.toISOString(), accountId }),
      ])

      console.log(chalk.bold(`\nReport: ${from.toISOString().slice(0, 10)} → ${to.toISOString().slice(0, 10)}`))
      console.log(`  Total in:  ${chalk.green('+' + summary.totalIn)}`)
      console.log(`  Total out: ${chalk.red('-' + summary.totalOut)}`)
      console.log(`  Net:       ${summary.net >= 0 ? chalk.green('+' + summary.net) : chalk.red(String(summary.net))}`)
      console.log(`  Transactions: ${summary.transactionCount}`)

      if (breakdown.length) {
        console.log(chalk.bold('\nSpend by category:'))
        breakdown.forEach(b => {
          console.log(`  ${b.categoryName.padEnd(20)} ${String(b.total).padStart(10)}`)
        })
      }
    })
}

function resolvePeriod(period: string): { from: Date; to: Date } {
  const now = new Date()
  if (period === 'this-month') {
    return {
      from: new Date(now.getFullYear(), now.getMonth(), 1),
      to: new Date(now.getFullYear(), now.getMonth() + 1, 0),
    }
  }
  if (period === 'this-week') {
    const day = now.getDay()
    const from = new Date(now.getTime() - day * 86_400_000)
    from.setHours(0, 0, 0, 0)
    return { from, to: new Date(from.getTime() + 7 * 86_400_000) }
  }
  // YYYY-MM format
  const [y, m] = period.split('-').map(Number)
  return {
    from: new Date(y, m - 1, 1),
    to: new Date(y, m, 0),
  }
}
```
